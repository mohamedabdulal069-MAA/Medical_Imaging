    .sample(n=64, random_state=SEED)

    "purpose": "Small pipeline pilot; not final research evaluation",
    "baseline": "resnet50",
    "initial_weights": "imagenet",
    "frozen_backbone": True,
    "seed": SEED,
    "training_images": len(train_rows),
    "validation_images": len(val_rows),

    "batch_size": BATCH_SIZE,

    "labels": list(LABELS),
    "test_images_used": 0,
    "tensorflow_version": tf.__version__,
}

            flush=True,
        )

    train_rows, (224, 224), BATCH_SIZE, training=True, seed=SEED
